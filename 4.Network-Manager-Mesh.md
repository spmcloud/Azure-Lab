## Overview 
 - Deploys three virtual networks and use Azure Virtual Network Manager to create a mesh network topology.
 - Network Manager gives a centralized place to manage Vnet connectivity 
 - A mesh network allow every Vnet to reach each other directly 

## Manual deployment

Create a resource group
``` bash
az group create --name RG-Lab-Mesh --location australiaeast
```

Create a network manager
``` bash
az network manager create \
    --location australiaeast \
    --name NM-Lab-Mesh \
    --resource-group RG-Lab-Mesh \
    --scope-accesses Connectivity SecurityAdmin \
    --network-manager-scopes subscriptions=/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b

```

Create a Network Group
``` bash
az network manager group create \
  --name network-group \
  --network-manager NM-Lab-Mesh \
  --resource-group RG-Lab-Mesh \
  --description "Network group for mesh lab VNets"
```

Create virtual networks
``` bash
az network vnet create \
  --name "vnet-00" \
  --resource-group "RG-Lab-Mesh" \
  --address-prefix "10.0.0.0/16" \
  --tags "NetworkType=Prod"

az network vnet create \
  --name "vnet-01" \
  --resource-group "RG-Lab-Mesh" \
  --address-prefix "10.1.0.0/16" \
  --tags "NetworkType=Prod"

az network vnet create \
  --name "vnet-02" \
  --resource-group "RG-Lab-Mesh" \
  --address-prefix "10.2.0.0/16" \
  --tags "NetworkType=Prod"
```

Add a subnet to each virtual network
``` bash
az network vnet subnet create \
--name "default" \
  --resource-group "RG-Lab-Mesh" \
  --vnet-name "vnet-00" \
  --address-prefix "10.0.0.0/24"

az network vnet subnet create \
  --name "default" \
  --resource-group "RG-Lab-Mesh" \
  --vnet-name "vnet-01" \
  --address-prefix "10.1.0.0/24"

az network vnet subnet create \
  --name "default" \
  --resource-group "RG-Lab-Mesh" \
  --vnet-name "vnet-02" \
  --address-prefix "10.2.0.0/24"
```

### Define membership for a mesh configuration
Network Manager allows you to add memberships dynamically or statically. Membership = which VNets Azure Network Manager should control.

To add this statically  
```bash
az network manager group static-member create \
  --name vnet-00 \
  --network-group network-group \
  --network-manager NM-Lab-Mesh \
  --resource-group RG-Lab-Mesh \
  --resource-id "/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b/resourceGroups/RG-Lab-Mesh/providers/Microsoft.Network/virtualNetworks/vnet-00"

az network manager group static-member create \
  --name vnet-01 \
  --network-group network-group \
  --network-manager NM-Lab-Mesh \
  --resource-group RG-Lab-Mesh \
  --resource-id "/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b/resourceGroups/RG-Lab-Mesh/providers/Microsoft.Network/virtualNetworks/vnet-01"

az network manager group static-member create \
  --name vnet-02 \
  --network-group network-group \
  --network-manager NM-Lab-Mesh \
  --resource-group RG-Lab-Mesh \
  --resource-id "/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b/resourceGroups/RG-Lab-Mesh/providers/Microsoft.Network/virtualNetworks/vnet-02"
```

Check membership has been created
``` bash
az network manager group static-member list \
  --network-manager NM-Lab-Mesh \
  --resource-group RG-Lab-Mesh \
  --network-group network-group \
  --output table
```

Create a mesh connectivity configuration
``` bash
az network manager connect-config create \
    --configuration-name "connectivityconfig" \
    --description "Mesh Connectivity Config for Lab" \
    --applies-to-groups '[{"networkGroupId": "/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b/resourceGroups/RG-Lab-Mesh/providers/Microsoft.Network/networkManagers/NM-Lab-Mesh/networkGroups/network-group", "groupConnectivity": "None"}]' \
    --connectivity-topology "Mesh" \
    --network-manager-name "NM-Lab-Mesh" \
    --resource-group "RG-Lab-Mesh"

```
- `groupConnectivity: "None"` - all VNets in the group are fully meshed.
- `connectivity-topology: "Mesh"` - traffic is allowed directly between all VNets.

Commit the configuration
``` bash
az network manager post-commit \
    --network-manager-name "NM-Lab-Mesh" \
    --commit-type "Connectivity" \
    --configuration-ids "/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b/resourceGroups/RG-Lab-Mesh/providers/Microsoft.Network/networkManagers/NM-Lab-Mesh/connectivityConfigurations/connectivityconfig" \
    --target-locations "australiaeast" \
    --resource-group "RG-Lab-Mesh"

```
- For the configuration to take effect, commit the configuration to the target regions

Verify the configuration
``` bash
az network manager list-effective-connectivity-config \
    --resource-group "RG-Lab-Mesh" \
    --virtual-network-name "vnet-00"

az network manager list-effective-connectivity-config \
    --resource-group "RG-Lab-Mesh" \
    --virtual-network-name "vnet-01"

az network manager list-effective-connectivity-config \
    --resource-group "RG-Lab-Mesh" \
    --virtual-network-name "vnet-02"

```
 - Output should show `connectivityTopology: "Mesh"` and `provisioningState: "Succeeded"`

Cleanup
``` bash
az group delete --name RG-Lab-Mesh --yes --no-wait
```



## Terraform deployment

```

 └── providers.tf
```

```
mkdir -p ~/terraform/3.network-manager-mesh/
```

### providers.tf
``` bash
cat << 'EOF' > ~/terraform/3.network-manager-mesh/providers.tf
terraform {
  required_version = ">= 1.5.0"

  required_providers {
    azurerm = {
      source  = "hashicorp/azurerm"
      version = "~> 4.0"
    }
  }
}

provider "azurerm" {
  features {}
  use_cli = true
  subscription_id = "6908ec90-4b9c-4100-aa00-4822de4f4e4b"
  resource_provider_registrations = "none"
}
EOF
```

### main.tf
``` bash
cat << 'EOF' > ~/terraform/3.network-manager-mesh/main.tf

# Variables

locals {
  subscription_id = "/subscriptions/6908ec90-4b9c-4100-aa00-4822de4f4e4b"

  vnets = {
    vnet-00 = { address = "10.0.0.0/16", subnet = "10.0.0.0/24" }
    vnet-01 = { address = "10.1.0.0/16", subnet = "10.1.0.0/24" }
    vnet-02 = { address = "10.2.0.0/16", subnet = "10.2.0.0/24" }
  }
}

# Create the resource group

resource "azurerm_resource_group" "rg" {
  name     = "RG-Lab-Mesh"
  location = "australiaeast"
}

# Create a Virtual Network Manager

resource "azurerm_network_manager" "nm" {
  name                = "NM-Lab-Mesh"
  resource_group_name = azurerm_resource_group.rg.name
  location            = azurerm_resource_group.rg.location

  scope_accesses = ["Connectivity", "SecurityAdmin"]

  scope {
    subscription_ids = [local.subscription_id]
  }
}

# Create a network group

resource "azurerm_network_manager_network_group" "ng" {
  name                = "network-group"
  network_manager_id  = azurerm_network_manager.nm.id
  description         = "Network group for mesh lab VNets"
}


# Create the Virtual Networks & Subnets

resource "azurerm_virtual_network" "vnet" {
  for_each            = local.vnets
  name                = each.key
  resource_group_name = azurerm_resource_group.rg.name
  location            = azurerm_resource_group.rg.location
  address_space       = [each.value.address]

  tags = {
    NetworkType = "Prod"
  }
}

resource "azurerm_subnet" "subnet" {
  for_each             = local.vnets
  name                 = "default"
  resource_group_name  = azurerm_resource_group.rg.name
  virtual_network_name = azurerm_virtual_network.vnet[each.key].name
  address_prefixes     = [each.value.subnet]
}


# Set the Static Membership

resource "azurerm_network_manager_static_member" "members" {
  for_each = azurerm_virtual_network.vnet

  name                       = each.key
  network_group_id           = azurerm_network_manager_network_group.ng.id
  target_virtual_network_id  = each.value.id
}

# Create a connectivity configuration

resource "azurerm_network_manager_connectivity_configuration" "mesh" {
  name                = "connectivityconfig"
  network_manager_id  = azurerm_network_manager.nm.id
  connectivity_topology = "Mesh"
  description         = "Mesh Connectivity Config for Lab"

  applies_to_group {
    network_group_id     = azurerm_network_manager_network_group.ng.id
    group_connectivity   = "None"
  }
}

# Commit deployment
resource "azurerm_network_manager_deployment" "commit_deployment" {
  network_manager_id = azurerm_network_manager.nm.id
  location           = azurerm_resource_group.rg.location
  scope_access       = "Connectivity"
  configuration_ids  = [azurerm_network_manager_connectivity_configuration.mesh.id]
}

EOF
```
 - Use `local.variable` if the variable is being referenced in the `main.tf` e.g. `[local.subscription_id]`
 - After you create a connectivity configuration, you need to commit it. It doesn't automatically apply to your network resources.
 - Management Groups were referenced in the MS learn page. These sit above subscriptions and are not required in this lab
