Modules are like ansible roles. They help reduce repetition

## 1. **Module Structure**

A typical module has its own folder and contains the resources you want to reuse, plus `variables.tf` and `outputs.tf`:

```
terraform/
 ├── modules/
 │   ├── vm/
 │   │   ├── main.tf
 │   │   └── variables.tf
 └── 3.modules-deploy/
     ├── main.tf
     ├── providers.tf
     └── variables.tf
```

```
mkdir -p ~/terraform/modules/vm/
```

```
cp ~/terraform/2.vm-lab/ ~/terraform/3.modules-deploy/
```

## VM Module

### `modules/vm/main.tf`

```
mkdir -p ~/terraform/modules/vm/
```

```
nano ~/terraform/modules/vm/main.tf
```

``` bash
# creates a NIC for the VM
resource "azurerm_network_interface" "nic" {
  name                = "${var.name}-nic"
  location            = var.location
  resource_group_name = var.resource_group_name

  ip_configuration {
    name                          = "internal"
    subnet_id                     = var.subnet_id
    private_ip_address_allocation = "Dynamic"
    public_ip_address_id          = var.public_ip_id != "" ? var.public_ip_id : null
  }
}


resource "azurerm_linux_virtual_machine" "vm" {
  name                = var.name
  location            = var.location
  resource_group_name = var.resource_group_name
  size                = "Standard_B1s"
  admin_username      = "azureadmin"
  network_interface_ids = [
    azurerm_network_interface.nic.id
  ]

  admin_ssh_key {
    username   = "azureadmin"
    public_key = file("~/.ssh/azure_key.pub")
  }

  os_disk {
    caching              = "ReadWrite"
    storage_account_type = "StandardSSD_LRS"
  }

  source_image_reference {
    publisher = "Canonical"
    offer     = "0001-com-ubuntu-server-jammy"
    sku       = "22_04-lts"
    version   = "latest"
  }

}
```
- If `var.public_ip_id` is empty (`""`), Terraform will assign `null` and **no public IP** will be attached.
- If `var.public_ip_id` contains an ID, the NIC will use it.

### `modules/vm/variables.tf`

```
nano ~/terraform/modules/vm/variables.tf
```

This defines all the variables your module needs
``` bash
variable "name" {}
variable "resource_group_name" {}
variable "location" {}
variable "subnet_id" {}
variable "public_ip_id" {}
```

### Using the VM Module
In `3.modules-deploy/main.tf`:

``` bash
module "jump_vm" {
  source              = "../modules/vm"
  name                = "vm-jump"
  resource_group_name = azurerm_resource_group.lab.name
  location            = azurerm_resource_group.lab.location
  subnet_id           = azurerm_subnet.subnet_jump.id
  public_ip_id        = azurerm_public_ip.jump_pip.id
}

module "web_vm" {
  source              = "../modules/vm"
  name                = "vm-web"
  resource_group_name = azurerm_resource_group.lab.name
  location            = azurerm_resource_group.lab.location
  subnet_id           = azurerm_subnet.subnet_web.id
  public_ip_id        = azurerm_public_ip.web_pip.id
}
```

  name     = "RG-Modules-Lab"

  name                = "vnet-lab"
